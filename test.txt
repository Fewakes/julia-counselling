/** Postgres connection (server-only). Works with Supabase's pooled connection strings, Neon, or a local Postgres. */
import postgres from "postgres";
import { connectionUrl } from "./db-url";

type Sql = ReturnType<typeof postgres>;

const globalForDb = globalThis as unknown as { juliaSql?: Sql };

export function databaseConfigured(): boolean {
  return Boolean(process.env.DATABASE_URL);
}

export function db(): Sql {
  const raw = process.env.DATABASE_URL;
  if (!raw) throw new Error("DATABASE_URL is not configured.");
  if (!globalForDb.juliaSql) {
    const url = connectionUrl(raw);
    if (url !== raw) console.warn("[db] DATABASE_URL uses Supabase's transaction pooler (port 6543); connecting through the session pooler (port 5432) instead.");
    const local = /@(localhost|127\.0\.0\.1)(:\d+)?\//.test(url) || process.env.DATABASE_SSL === "false";
    globalForDb.juliaSql = postgres(url, {
      prepare: false,
      // Few connections per server instance: the session pooler keeps one database connection per client connection.
      max: 3,
      // Close idle connections quickly, so a paused serverless instance does not hold stale sockets or pooler slots.
      idle_timeout: 10,
      connect_timeout: 10,
      ssl: local ? false : "require",
    });
  }
  return globalForDb.juliaSql;
}

/** Postgres error code helper (e.g. 23P01 = exclusion violation, 23505 = unique violation). */
export function pgErrorCode(error: unknown): string | null {
  if (error && typeof error === "object" && "code" in error) {
    const code = (error as { code?: unknown }).code;
    return typeof code === "string" ? code : null;
  }
  return null;
}
